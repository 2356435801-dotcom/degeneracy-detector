AdaptiveWeightCalculator::
AdaptiveWeightCalculator()
{
    translation_max_=10.0;

    rotation_weight_=40.0;

    gain_=0.03;
}

void AdaptiveWeightCalculator::
setMaximumTranslation(double w)
{
    translation_max_=w;
}

void AdaptiveWeightCalculator::
setRotationWeight(double w)
{
    rotation_weight_=w;
}

void AdaptiveWeightCalculator::
setGain(double k)
{
    gain_=k;
}

WeightResult
AdaptiveWeightCalculator::
calculate(double score) const
{
    WeightResult result;

    result.translation_weight=
            translation_max_/
            (1.0+gain_*score);

    result.rotation_weight=
            rotation_weight_;

    return result;
}

bool CovarianceCalculator::compute(
        const ScanFrame& frame,
        CovarianceResult& result) const
{
    // 至少需要两个点
    if(frame.points.size() < 2)
    {
        return false;
    }

    //----------------------------------------------------------------------
    // Step 1 : 计算质心
    //----------------------------------------------------------------------

    result.centroid.setZero();

    for(const auto& point : frame.points)
    {
        result.centroid += point;
    }

    result.centroid /= static_cast<double>(frame.points.size());

    //----------------------------------------------------------------------
    // Step 2 : 计算协方差矩阵
    //----------------------------------------------------------------------

    result.covariance.setZero();

    for(const auto& point : frame.points)
    {
        Eigen::Vector2d diff = point - result.centroid;

        result.covariance += diff * diff.transpose();
    }

    result.covariance /=
            static_cast<double>(frame.points.size() - 1);

    //----------------------------------------------------------------------
    // Step 3 : PCA（特征值分解）
    //----------------------------------------------------------------------

    Eigen::SelfAdjointEigenSolver<Eigen::Matrix2d> solver(
            result.covariance);

    if(solver.info() != Eigen::Success)
    {
        return false;
    }

    result.eigen_values =
            solver.eigenvalues();

    result.eigen_vectors =
            solver.eigenvectors();

    return true;

DegeneracyEngine::DegeneracyEngine()
{
}

bool DegeneracyEngine::update(
        const sensor_msgs::LaserScan& scan,
        ScanFrame& merged_frame,
        CovarianceResult& covariance_result,
        DegeneracyResult& degeneracy_result,
        WeightResult& weight_result)
{
    //----------------------------------
    // Step1
    //----------------------------------
    if(!point_processor_.convertLaserScan(scan,
                                          current_frame_))
    {
        return false;
    }

    //----------------------------------
    // Step2
    //----------------------------------
    sliding_window_.addFrame(current_frame_);

    merged_frame = sliding_window_.mergedFrame();

    //----------------------------------
    // Step3
    //----------------------------------
    if(!covariance_calculator_.compute(
            merged_frame,
            covariance_result))
    {
        return false;
    }

    //----------------------------------
    // Step4
    //----------------------------------
    if(!analyzer_.analyze(
            covariance_result,
            degeneracy_result))
    {
        return false;
    }

    //----------------------------------
    // Step5
    //----------------------------------
    weight_result =
            weight_calculator_.calculate(
                    degeneracy_result.score);

    return true;
}

PointProcessor::PointProcessor()
{
    min_range_ = 0.10;
    max_range_ = 30.0;
}

void PointProcessor::setRangeLimit(
        double min_range,
        double max_range)
{
    min_range_ = min_range;
    max_range_ = max_range;
}

double PointProcessor::minRange() const
{
    return min_range_;
}

double PointProcessor::maxRange() const
{
    return max_range_;
}

bool PointProcessor::convertLaserScan(
        const sensor_msgs::LaserScan& scan,
        ScanFrame& frame)
{
    // 清空上一帧数据
    frame.points.clear();

    // 保存时间戳
    frame.stamp = scan.header.stamp;

    // 预分配空间，减少vector扩容次数
    frame.points.reserve(scan.ranges.size());

    for(size_t i = 0; i < scan.ranges.size(); ++i)
    {
        double range = scan.ranges[i];

        // NaN过滤
        if(std::isnan(range))
            continue;

        // Inf过滤
        if(std::isinf(range))
            continue;

        // 最小距离过滤
        if(range < min_range_)
            continue;

        // 最大距离过滤
        if(range > max_range_)
            continue;

        double angle =
                scan.angle_min +
                i * scan.angle_increment;

        double x = range * std::cos(angle);

        double y = range * std::sin(angle);

        frame.points.emplace_back(x, y);
    }

    // 点数过少直接认为失败
    if(frame.points.size() < 10)
        return false;

    return true;
}

#include "sliding_window.h"

SlidingWindow::SlidingWindow(size_t window_size)
    : window_size_(window_size)
{
}

void SlidingWindow::addFrame(const ScanFrame& frame)
{
    window_.push_back(frame);

    while(window_.size() > window_size_)
    {
        window_.pop_front();
    }

    rebuildMergedFrame();
}

void SlidingWindow::rebuildMergedFrame()
{
    merged_frame_.points.clear();

    if(window_.empty())
        return;

    // 使用最新帧时间戳
    merged_frame_.stamp = window_.back().stamp;

    // 统计总点数，提前分配内存
    size_t total_points = 0;

    for(const auto& frame : window_)
    {
        total_points += frame.points.size();
    }

    merged_frame_.points.reserve(total_points);

    // 合并点云
    for(const auto& frame : window_)
    {
        merged_frame_.points.insert(
                merged_frame_.points.end(),
                frame.points.begin(),
                frame.points.end());
    }
}

const ScanFrame& SlidingWindow::mergedFrame() const
{
    return merged_frame_;
}

size_t SlidingWindow::frameCount() const
{
    return window_.size();
}

size_t SlidingWindow::windowSize() const
{
    return window_size_;
}
